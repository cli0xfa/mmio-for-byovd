cmake_minimum_required(VERSION 3.20)
project(mmio_for_byovd LANGUAGES CXX)

option(MMIO_BUILD_TESTS "Build driver-free model tests" ON)
if(MMIO_BUILD_TESTS)
    enable_testing()
    add_executable(mmio_paging_model_tests tests/test_paging_model.cpp)
    target_compile_features(mmio_paging_model_tests PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(mmio_paging_model_tests PRIVATE /W4 /permissive-)
    else()
        target_compile_options(mmio_paging_model_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME mmio_paging_model COMMAND mmio_paging_model_tests)

    if(WIN32)
        add_executable(mmio_windows_header_smoke tests/windows_header_smoke.cpp)
        target_compile_features(mmio_windows_header_smoke PRIVATE cxx_std_20)
        target_include_directories(mmio_windows_header_smoke PRIVATE mmio-for-byovd)
        target_link_libraries(mmio_windows_header_smoke PRIVATE psapi advapi32)
        if(MSVC)
            target_compile_options(mmio_windows_header_smoke PRIVATE /W4 /permissive- /EHsc /analyze)
        endif()
        add_test(NAME mmio_windows_header_smoke COMMAND mmio_windows_header_smoke)
    endif()
endif()
